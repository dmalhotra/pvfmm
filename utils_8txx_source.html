7template <class FMM_Mat_t>
8void CheckFMMOutput(pvfmm::FMM_Tree<FMM_Mat_t>* mytree, const pvfmm::Kernel<typename FMM_Mat_t::Real_t>* mykernel, std::string t_name){
9 if(mykernel==NULL) return;
10
11 // Find out number of OMP thereads.
12 int np=omp_get_max_threads();
13
14 // Find out my identity in the default communicator
15 const sctl::Comm c1 = sctl::Comm::World();
16 const int myrank = c1.Rank();
17 const int p = c1.Size();
18
19 //typedef typename FMM_Mat_t::FMMData FMM_Data_t;
20 typedef typename FMM_Mat_t::FMMNode_t FMMNode_t;
21 typedef typename FMM_Mat_t::Real_t Real_t;
22
23 // Read source data.
24 std::vector<Real_t> src_coord;
25 std::vector<Real_t> src_value;
26 sctl::Iterator<FMMNode_t> n=mytree->PreorderFirst();
27 while(n!=sctl::NullIterator<FMMNode_t>()){
28 if(n->IsLeaf() && !n->IsGhost()){
29 sctl::Vector<Real_t>& coord_vec=n->src_coord;
30 sctl::Vector<Real_t>& value_vec=n->src_value;
31 for(sctl::Long i=0;i<coord_vec.Dim();i++) src_coord.push_back(coord_vec[i]);
32 for(sctl::Long i=0;i<value_vec.Dim();i++) src_value.push_back(value_vec[i]);
33 }
34 n=mytree->PreorderNxt(n);
35 }
36 long long glb_src_cnt=0, src_cnt=src_coord.size()/3;
37 c1.Allreduce(sctl::Ptr2ConstItr<long long>(&src_cnt, 1), sctl::Ptr2Itr<long long>(&glb_src_cnt, 1), 1, sctl::CommOp::SUM);
38 long long glb_val_cnt=0, val_cnt=src_value.size();
39 c1.Allreduce(sctl::Ptr2ConstItr<long long>(&val_cnt, 1), sctl::Ptr2Itr<long long>(&glb_val_cnt, 1), 1, sctl::CommOp::SUM);
40 if(glb_src_cnt==0) return;
41
42 int dof=glb_val_cnt/glb_src_cnt/mykernel->ker_dim[0];
43 int trg_dof=dof*mykernel->ker_dim[1];
44
45 // Read target data.
46 std::vector<Real_t> trg_coord;
47 std::vector<Real_t> trg_poten_fmm;
48 long long trg_iter=0;
49 size_t step_size=(size_t)(1+glb_src_cnt*glb_src_cnt*1e-9/p);
50 n=mytree->PreorderFirst();
51 while(n!=sctl::NullIterator<FMMNode_t>()){
52 if(n->IsLeaf() && !n->IsGhost()){
53 sctl::Vector<Real_t>& coord_vec=n->trg_coord;
54 sctl::Vector<Real_t>& poten_vec=n->trg_value;
55 for(sctl::Long i=0;i<coord_vec.Dim()/3 ;i++){
56 if(trg_iter%step_size==0){
57 for(int j=0;j<3 ;j++) trg_coord .push_back(coord_vec[i*3 +j]);
58 for(int j=0;j<trg_dof ;j++) trg_poten_fmm.push_back(poten_vec[i*trg_dof +j]);
59 }
60 trg_iter++;
61 }
62 }
63 n=mytree->PreorderNxt(n);
64 }
65 int trg_cnt=trg_coord.size()/3;
66 int send_cnt=trg_cnt*3;
67 std::vector<sctl::Long> recv_cnts(p), recv_disp(p, 0);
68 c1.Allgather(sctl::Ptr2ConstItr<int>(&send_cnt, 1), 1,
69 sctl::Ptr2Itr<sctl::Long>(&recv_cnts[0], p), 1);
70 sctl::omp_par::scan(&recv_cnts[0], &recv_disp[0], p);
71 int glb_trg_cnt = (recv_disp[p-1] + recv_cnts[p-1]) / 3;
72 std::vector<Real_t> glb_trg_coord(glb_trg_cnt * 3);
73 c1.Allgatherv(sctl::Ptr2ConstItr<Real_t>(&trg_coord[0], send_cnt), send_cnt,
74 sctl::Ptr2Itr<Real_t>(&glb_trg_coord[0], glb_trg_cnt * 3),
75 sctl::Ptr2ConstItr<sctl::Long>(&recv_cnts[0], p),
76 sctl::Ptr2ConstItr<sctl::Long>(&recv_disp[0], p));
77 if(glb_trg_cnt==0) return;
78
79 //Direct N-Body.
80 std::vector<Real_t> trg_poten_dir(glb_trg_cnt*trg_dof ,0);
81 std::vector<Real_t> glb_trg_poten_dir(glb_trg_cnt*trg_dof ,0);
82 sctl::Profile::Tic("N-Body Direct", &c1, false, 1);
83 #pragma omp parallel for
84 for(int i=0;i<np;i++){
85 size_t a=(i*glb_trg_cnt)/np;
86 size_t b=((i+1)*glb_trg_cnt)/np;
87 mykernel->ker_poten(&src_coord[0], src_cnt, &src_value[0], dof, &glb_trg_coord[a*3], b-a, &trg_poten_dir[a*trg_dof ]);
88 }
89 c1.Allreduce(sctl::Ptr2ConstItr<Real_t>(&trg_poten_dir[0], trg_poten_dir.size()),
90 sctl::Ptr2Itr<Real_t>(&glb_trg_poten_dir[0], glb_trg_poten_dir.size()),
91 (sctl::Long)trg_poten_dir.size(), sctl::CommOp::SUM);
92 sctl::Profile::Toc();
93
94 //Compute error.
95 {
96 Real_t max_=0;
97 Real_t max_err=0;
98 for(size_t i=0;i<trg_poten_fmm.size();i++){
99 Real_t err=(Real_t)fabs(glb_trg_poten_dir[i+(recv_disp[myrank]/3)*trg_dof]-trg_poten_fmm[i]);
100 Real_t max=(Real_t)fabs(glb_trg_poten_dir[i+(recv_disp[myrank]/3)*trg_dof]);
101 if(err>max_err) max_err=err;
102 if(max>max_) max_=max;
103 }
104 Real_t glb_max, glb_max_err;
105 c1.Allreduce(sctl::Ptr2ConstItr<Real_t>(&max_, 1), sctl::Ptr2Itr<Real_t>(&glb_max, 1), 1, sctl::CommOp::MAX);
106 c1.Allreduce(sctl::Ptr2ConstItr<Real_t>(&max_err, 1), sctl::Ptr2Itr<Real_t>(&glb_max_err, 1), 1, sctl::CommOp::MAX);
107 if(!myrank){
108 std::cout<<"Maximum Absolute Error ["<<t_name<<"] : "<<std::scientific<<glb_max_err<<'\n';
109 std::cout<<"Maximum Relative Error ["<<t_name<<"] : "<<std::scientific<<glb_max_err/glb_max<<'\n';
110 }
111 }
112}

283template <class Real_t>
284std::vector<Real_t> point_distrib(DistribType dist_type, size_t N, const sctl::Comm& comm){
285 const int np = comm.Size();
286 const int myrank = comm.Rank();
287 static size_t seed=myrank+1; seed+=np;
288 srand48(seed);
289
290 std::vector<Real_t> coord;
291 switch(dist_type){
292 case UnifGrid:
293 {
294 size_t NN=(size_t)round(pow((double)N,1.0/3.0));
295 size_t N_total=NN*NN*NN;
296 size_t start= myrank *N_total/np;
297 size_t end =(myrank+1)*N_total/np;
298 for(size_t i=start;i<end;i++){
299 coord.push_back((Real_t)(((i/ 1 )%NN)+0.5)/NN);
300 coord.push_back((Real_t)(((i/ NN )%NN)+0.5)/NN);
301 coord.push_back((Real_t)(((i/(NN*NN))%NN)+0.5)/NN);
302 }
303 }
304 break;
305 case RandUnif:
306 {
307 size_t N_total=N;
308 size_t start= myrank *N_total/np;
309 size_t end =(myrank+1)*N_total/np;
310 size_t N_local=end-start;
311 coord.resize(N_local*3);
312
313 for(size_t i=0;i<N_local*3;i++)
314 coord[i]=((Real_t)drand48());
315 }
316 break;
317 case RandGaus:
318 {
319 Real_t e=sctl::const_e<Real_t>();
320 Real_t log_e=sctl::log(e);
321 size_t N_total=N;
322 size_t start= myrank *N_total/np;
323 size_t end =(myrank+1)*N_total/np;
324
325 for(size_t i=start*3;i<end*3;i++){
326 Real_t y=-1;
327 while(y<=0 || y>=1){
328 Real_t r1=(Real_t)(sqrt(-2*log(drand48())/log_e)*cos(2*M_PI*drand48()));
329 Real_t r2=(Real_t)pow(0.5,i*10/N_total);
330 y=(Real_t)0.5+r1*r2;
331 }
332 coord.push_back(y);
333 }
334 }
335 break;
336 case RandElps:
337 {
338 size_t N_total=N;
339 size_t start= myrank *N_total/np;
340 size_t end =(myrank+1)*N_total/np;
341 size_t N_local=end-start;
342 coord.resize(N_local*3);
343
344 const Real_t r=(Real_t)0.45;
345 const Real_t center[3]={0.5,0.5,0.5};
346 for(size_t i=0;i<N_local;i++){
347 Real_t* y=&coord[i*3];
348 Real_t phi=(Real_t)(2*M_PI*drand48());
349 Real_t theta=(Real_t)(M_PI*drand48());
350 y[0]=center[0]+(Real_t)(0.25*r*sin(theta)*cos(phi));
351 y[1]=center[1]+(Real_t)(0.25*r*sin(theta)*sin(phi));
352 y[2]=center[2]+r*(Real_t)cos(theta);
353 }
354 }
355 break;
356 case RandSphr:
357 {
358 size_t N_total=N;
359 size_t start= myrank *N_total/np;
360 size_t end =(myrank+1)*N_total/np;
361 size_t N_local=end-start;
362 coord.resize(N_local*3);
363
364 const Real_t center[3]={0.5,0.5,0.5};
365 for(size_t i=0;i<N_local;i++){
366 Real_t* y=&coord[i*3];
367 Real_t r=1;
368 while(r>0.5 || r==0){
369 y[0]=(Real_t)drand48();
370 y[1]=(Real_t)drand48();
371 y[2]=(Real_t)drand48();
372 r=(Real_t)sqrt((y[0]-center[0])*(y[0]-center[0])
373 +(y[1]-center[1])*(y[1]-center[1])
374 +(y[2]-center[2])*(y[2]-center[2]));
375 y[0]=center[0]+(Real_t)0.45*(y[0]-center[0])/r;
376 y[1]=center[1]+(Real_t)0.45*(y[1]-center[1])/r;
377 y[2]=center[2]+(Real_t)0.45*(y[2]-center[2])/r;
378 }
379 }
380 }
381 break;
382 default:
383 break;
384 }
385 return coord;
386}
