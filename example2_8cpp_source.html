44 // Run FMM
45 std::vector<double> trg_value;
46 size_t n_trg=trg_coord.size()/PVFMM_COORD_DIM;
47 pvfmm::ChebFMM_Evaluate(trg_value, tree, n_trg);
48
49 // Re-run FMM
50 tree->ClearFMMData();
51 pvfmm::ChebFMM_Evaluate(trg_value, tree, n_trg);
52
53 {// Check error
54 std::vector<double> trg_value_(n_trg*kernel_fn.ker_dim[1]);
55 fn_output(&trg_coord[0],n_trg,&trg_value_[0]);
56 double max_err=0;
57 double max_err_glb=0;
58 for(size_t i=0;i<n_trg;i++){
59 if(fabs(trg_value_[i]-trg_value[i])>max_err)
60 max_err=fabs(trg_value_[i]-trg_value[i]);
61 }
62 comm.Allreduce(sctl::Ptr2ConstItr<double>(&max_err, 1),
63 sctl::Ptr2Itr<double>(&max_err_glb, 1), 1, sctl::CommOp::MAX);
64 if (!comm.Rank()) std::cout<<"Maximum Error:"<<max_err_glb<<'\n';
65 }

67 // Free memory
68 delete tree;
