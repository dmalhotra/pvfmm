11void BiotSavart(const double* src_X, const double* src_V, long Ns, const double* trg_X, double* trg_V, long Nt) { // Direct evaluation
12 double oofp = 1 / (4 * M_PI);
13 #pragma omp parallel for schedule(static)
14 for (long t = 0; t < Nt; t++) {
15 trg_V[t*3+0] = 0;
16 trg_V[t*3+1] = 0;
17 trg_V[t*3+2] = 0;
18 for (long s = 0; s < Ns; s++) {
19 double X[3];
20 X[0] = trg_X[t*3+0] - src_X[s*3+0];
21 X[1] = trg_X[t*3+1] - src_X[s*3+1];
22 X[2] = trg_X[t*3+2] - src_X[s*3+2];
23 double r2 = X[0]*X[0] + X[1]*X[1] + X[2]*X[2];
24 double rinv = (r2 > 0 ? 1/sqrt(r2) : 0);
25 double rinv3 = rinv * rinv * rinv;
26
27 trg_V[t*3+0] += (src_V[s*3+1]*X[2] - src_V[s*3+2]*X[1]) * rinv3 * oofp;
28 trg_V[t*3+1] += (src_V[s*3+2]*X[0] - src_V[s*3+0]*X[2]) * rinv3 * oofp;
29 trg_V[t*3+2] += (src_V[s*3+0]*X[1] - src_V[s*3+1]*X[0]) * rinv3 * oofp;
30 }
31 }
32}

34void test_FMM(void* ctx) { // Compare FMM and direct evaluation results
35 int kdim[2] = {3,3};
36 long Ns = 20000;
37 long Nt = 20000;
38
39 // Initialize target coordinates
40 double* trg_X = (double*)malloc(Nt * 3 * sizeof(double));
41 double* trg_V = (double*)malloc(Nt * kdim[1] * sizeof(double));
42 double* trg_V0 = (double*)malloc(Nt * kdim[1] * sizeof(double));
43 for (long i = 0; i < Nt * 3; i++) trg_X[i] = drand48();
44
45 // Initialize source coordinates and density
46 double* src_X = (double*)malloc(Ns * 3 * sizeof(double));
47 double* src_V = (double*)malloc(Ns * kdim[0] * sizeof(double));
48 for (long i = 0; i < Ns * 3; i++) src_X[i] = drand48();
49 for (long i = 0; i < Ns * kdim[0]; i++) src_V[i] = drand48() - 0.5;
50
51 // FMM evaluation
52 double tt;
53 int setup = 1;
54 tt = -omp_get_wtime();
55 PVFMMEvalD(src_X, src_V, NULL, Ns, trg_X, trg_V, Nt, ctx, setup);
56 printf("FMM evaluation time (with setup) : %f\n", tt + omp_get_wtime());
57
58 // FMM evaluation (without setup)
59 setup = 0;
60 tt = -omp_get_wtime();
61 PVFMMEvalD(src_X, src_V, NULL, Ns, trg_X, trg_V, Nt, ctx, setup);
62 printf("FMM evaluation time (without setup) : %f\n", tt + omp_get_wtime());
63
64 // Direct evaluation
65 tt = -omp_get_wtime();
66 BiotSavart(&src_X[0], &src_V[0], Ns, &trg_X[0], &trg_V0[0], Nt);
67 printf("Direct evaluation time : %f\n", tt + omp_get_wtime());
68
69 double max_err = 0, max_val = 0;
70 for (long i = 0; i < Nt * kdim[1]; i++) { // Compute error
71 double val = fabs(trg_V0[i]), err = fabs(trg_V[i] - trg_V0[i]);
72 if (val > max_val) max_val = val;
73 if (err > max_err) max_err = err;
74 }
75 printf("Max relative error : %e\n", max_err / max_val);
76
77 free(src_X);
78 free(src_V);
79 free(trg_X);
80 free(trg_V);
81 free(trg_V0);
82}
