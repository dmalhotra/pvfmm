47 inline uintptr_t alloc_device_cuda(char* dev_handle, size_t len) {
48 char *dev_ptr=NULL;
49#if defined(PVFMM_HAVE_CUDA)
50 cudaError_t error;
51 error = cudaHostRegister(dev_handle, len, cudaHostRegisterPortable);
52 if (error != cudaSuccess)
53 std::cout<<cudaGetErrorString(error)<< '\n';
54 assert(error == cudaSuccess);
55 if(len) error = cudaMalloc((void**)&dev_ptr, len);
56 if (error != cudaSuccess)
57 std::cout<<cudaGetErrorString(error)<< '\n';
58 assert(error == cudaSuccess);
59#else
60 PVFMM_UNUSED(dev_handle);
61 PVFMM_UNUSED(len);
62#endif
63 return (uintptr_t)dev_ptr;
64 }

191 template <class T>
192 inline DeviceVector<T> DeviceMirror::AllocDevice(sctl::Vector<T>& host, bool copy){
193 char* p=(char*)(host.Dim()?&host[0]:nullptr);
194 size_t bytes=host.Dim()*sizeof(T);
195 if(dev_ptr){ // Already bound: host buffer must not have changed.
196 assert(host_ptr==p && len==bytes);
197 }else if(bytes){
198 host_ptr=p;
199 len=bytes;
200 dev_ptr=DeviceWrapper::alloc_device(host_ptr,len);
201 }
202 if(dev_ptr && copy) lock_idx=DeviceWrapper::host2device(host_ptr,host_ptr,dev_ptr,len);
203
204 DeviceVector<T> h;
205 h.dim=host.Dim();
206 h.dev_ptr=dev_ptr;
207 return h;
208 }

210 template <class T>
211 inline DeviceMatrix<T> DeviceMirror::AllocDevice(sctl::Matrix<T>& host, bool copy){
212 char* p=(char*)(host.Dim(0)*host.Dim(1)>0?&host[0][0]:nullptr);
213 size_t bytes=host.Dim(0)*host.Dim(1)*sizeof(T);
214 if(dev_ptr){ // Already bound: host buffer must not have changed.
215 assert(host_ptr==p && len==bytes);
216 }else if(bytes){
217 host_ptr=p;
218 len=bytes;
219 dev_ptr=DeviceWrapper::alloc_device(host_ptr,len);
220 }
221 if(dev_ptr && copy) lock_idx=DeviceWrapper::host2device(host_ptr,host_ptr,dev_ptr,len);
222
223 DeviceMatrix<T> h;
224 h.dim[0]=host.Dim(0);
225 h.dim[1]=host.Dim(1);
226 h.dev_ptr=dev_ptr;
227 h.lock_idx=lock_idx;

253 inline void CUDA_Lock::init(size_t num_stream) {
254 assert(num_stream>0);
255 if(num_stream==stream.size()) return;
256 cublasStatus_t status;
257 cudaError_t error;
258
259 // Delete previous streams
260 for(size_t i=0;i<stream.size();i++){
261 error = cudaStreamDestroy(stream[i]);
262 }
263
264 // Create new streams
265 stream.resize(num_stream);
266 for (size_t i = 0; i < num_stream; i++) {
267 error = cudaStreamCreate(&(stream[i]));
268 }
269
270 // Create cuBLAS context
271 static bool cuda_init=false;
272 if (!cuda_init) {
273 cuda_init = true;
274 status = cublasCreate(&handle);
275 }
276
277 // Set cuBLAS to use stream[0]
278 status = cublasSetStream(handle, stream[0]);
279 PVFMM_UNUSED(status);
280 PVFMM_UNUSED(error);
281 }
282
283 inline void CUDA_Lock::finalize () {
284 if(stream.size()==0) return;
285 for (size_t i = 0; i < stream.size(); i++) {
286 cudaError_t error = cudaStreamDestroy(stream[i]);
287 PVFMM_UNUSED(error);
288 }
289 cublasStatus_t status = cublasDestroy(handle);
290 PVFMM_UNUSED(status);
291 }
292
293 inline cudaStream_t *CUDA_Lock::acquire_stream (int idx) {
294 if (stream.size()<=(size_t)idx) init(idx+1);
295 return &(stream[idx]);
296 }
297
298 inline cublasHandle_t *CUDA_Lock::acquire_handle () {
299 if (stream.size()==0) init();
300 return &handle;
301 }
302
303 inline void CUDA_Lock::wait (int idx) {
304 if (stream.size()<=(size_t)idx) init(idx+1);
305 cudaError_t error = cudaStreamSynchronize(stream[idx]);
306 PVFMM_UNUSED(error);
307 }
