41template <class T>
42void MatrixTranspose(size_t in_dim1, size_t in_dim2, sctl::ConstIterator<T> in, sctl::Iterator<T> out){
43 constexpr size_t B1=128, B2=32; // cache-blocking tile sizes
44 const size_t d0=in_dim1;
45 const size_t d1=in_dim2;
46 if(d0*d1==0) return;
47
48 auto transpose_=[d0,d1](sctl::ConstIterator<T> in_, sctl::Iterator<T> out_){
49 const size_t blk0=((d0+B1-1)/B1);
50 const size_t blk1=((d1+B1-1)/B1);
51 for(size_t k=0;k<blk0*blk1;k++){
52 size_t i=(k%blk0)*B1;
53 size_t j=(k/blk0)*B1;
54 size_t d0_=i+B1; if(d0_>=d0) d0_=d0;
55 size_t d1_=j+B1; if(d1_>=d1) d1_=d1;
56 for(size_t ii=i;ii<d0_;ii+=B2)
57 for(size_t jj=j;jj<d1_;jj+=B2){
58 size_t d0__=ii+B2; if(d0__>=d0) d0__=d0;
59 size_t d1__=jj+B2; if(d1__>=d1) d1__=d1;
60 for(size_t iii=ii;iii<d0__;iii++)
61 for(size_t jjj=jj;jjj<d1__;jjj++){
62 out_[jjj*d0+iii]=in_[iii*d1+jjj];
63 }
64 }
65 }
66 };
67
68 if(in==(sctl::ConstIterator<T>)out){ // in-place: stage the input in scratch
69 sctl::ScratchBuf<T> buff((sctl::Long)(d0*d1));
70 sctl::Iterator<T> tmp=buff.begin();
71 sctl::omp_par::copy(in, in+(sctl::Long)(d0*d1), tmp);
72 transpose_((sctl::ConstIterator<T>)tmp, out);
73 }else{
74 transpose_(in, out);
75 }
76}
